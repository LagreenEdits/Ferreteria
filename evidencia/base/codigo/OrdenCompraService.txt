   1: package com.xorr.compras.service;
   2: 
   3: import java.math.BigDecimal;
   4: import java.time.LocalDateTime;
   5: import java.util.List;
   6: 
   7: import org.springframework.stereotype.Service;
   8: import org.springframework.transaction.annotation.Transactional;
   9: 
  10: import com.xorr.common.ContextoActual;
  11: import com.xorr.common.OperacionInvalidaException;
  12: import com.xorr.common.RecursoNoEncontradoException;
  13: import com.xorr.compras.dto.DetalleOrdenCompraRequest;
  14: import com.xorr.compras.dto.OrdenCompraRequest;
  15: import com.xorr.compras.dto.OrdenCompraResponse;
  16: import com.xorr.compras.entity.DetalleOrdenCompra;
  17: import com.xorr.compras.entity.EstadoOrdenCompra;
  18: import com.xorr.compras.entity.OrdenCompra;
  19: import com.xorr.compras.repository.OrdenCompraRepository;
  20: import com.xorr.inventario.service.ProductoService;
  21: import com.xorr.inventario.service.ProveedorProductoService;
  22: import com.xorr.inventario.service.ProveedorService;
  23: 
  24: @Service
  25: public class OrdenCompraService {
  26: 
  27:     private final OrdenCompraRepository ordenes;
  28:     private final ContextoActual contexto;
  29:     private final ProveedorService proveedorService;
  30:     private final ProductoService productoService;
  31:     private final ProveedorProductoService proveedorProductoService;
  32: 
  33:     public OrdenCompraService(OrdenCompraRepository ordenes,
  34:             ContextoActual contexto,
  35:             ProveedorService proveedorService,
  36:             ProductoService productoService,
  37:             ProveedorProductoService proveedorProductoService) {
  38:         this.ordenes = ordenes;
  39:         this.contexto = contexto;
  40:         this.proveedorService = proveedorService;
  41:         this.productoService = productoService;
  42:         this.proveedorProductoService = proveedorProductoService;
  43:     }
  44: 
  45:     @Transactional
  46:     public OrdenCompraResponse crear(OrdenCompraRequest datos) {
  47:         proveedorService.obtener(datos.idProveedor());
  48:         OrdenCompra orden = nuevaOrdenPendiente(datos.idProveedor());
  49:         datos.detalles().forEach(linea -> orden.getDetalles().add(armarDetalle(datos.idProveedor(), linea)));
  50:         orden.setTotal(sumarSubtotales(orden));
  51:         return OrdenCompraResponse.desde(ordenes.save(orden));
  52:     }
  53: 
  54:     @Transactional(readOnly = true)
  55:     public List<OrdenCompraResponse> listar() {
  56:         return ordenes.findByIdEmpresa(contexto.getIdEmpresa()).stream()
  57:                 .map(OrdenCompraResponse::desde)
  58:                 .toList();
  59:     }
  60: 
  61:     @Transactional(readOnly = true)
  62:     public OrdenCompraResponse obtener(Long idOrden) {
  63:         return OrdenCompraResponse.desde(buscarOrden(idOrden));
  64:     }
  65: 
  66:     // RN-02 y RN-06: si falla el aumento de stock de una línea, la transacción se
  67:     // revierte completa.
  68:     @Transactional
  69:     public OrdenCompraResponse recibir(Long idOrden) {
  70:         OrdenCompra orden = buscarOrden(idOrden);
  71:         validarPendiente(orden, "recibir");
  72:         orden.getDetalles().forEach(
  73:                 detalle -> productoService.incrementarStock(detalle.getIdProducto(), detalle.getCantidad()));
  74:         orden.setEstado(EstadoOrdenCompra.RECIBIDA);
  75:         return OrdenCompraResponse.desde(ordenes.save(orden));
  76:     }
  77: 
  78:     @Transactional
  79:     public OrdenCompraResponse cancelar(Long idOrden) {
  80:         OrdenCompra orden = buscarOrden(idOrden);
  81:         validarPendiente(orden, "cancelar");
  82:         orden.setEstado(EstadoOrdenCompra.CANCELADA);
  83:         return OrdenCompraResponse.desde(ordenes.save(orden));
  84:     }
  85: 
  86:     private OrdenCompra nuevaOrdenPendiente(Long idProveedor) {
  87:         OrdenCompra orden = new OrdenCompra();
  88:         orden.setIdEmpresa(contexto.getIdEmpresa());
  89:         orden.setIdProveedor(idProveedor);
  90:         orden.setIdUsuario(contexto.getIdUsuario());
  91:         orden.setFecha(LocalDateTime.now());
  92:         orden.setEstado(EstadoOrdenCompra.PENDIENTE);
  93:         orden.setTotal(BigDecimal.ZERO);
  94:         return orden;
  95:     }
  96: 
  97:     // RN-05 (producto de otra empresa = no encontrado), RN-09 (precio de la
  98:     // asociación)
  99:     // y RN-03 (el subtotal lo calcula el servidor).
 100:     private DetalleOrdenCompra armarDetalle(Long idProveedor, DetalleOrdenCompraRequest linea) {
 101:         productoService.obtener(linea.idProducto());
 102:         BigDecimal precio = proveedorProductoService.obtenerPrecioCompra(idProveedor, linea.idProducto());
 103:         DetalleOrdenCompra detalle = new DetalleOrdenCompra();
 104:         detalle.setIdProducto(linea.idProducto());
 105:         detalle.setCantidad(linea.cantidad());
 106:         detalle.setPrecioUnitario(precio);
 107:         detalle.setSubtotal(precio.multiply(BigDecimal.valueOf(linea.cantidad())));
 108:         return detalle;
 109:     }
 110: 
 111:     private BigDecimal sumarSubtotales(OrdenCompra orden) {
 112:         return orden.getDetalles().stream()
 113:                 .map(DetalleOrdenCompra::getSubtotal)
 114:                 .reduce(BigDecimal.ZERO, BigDecimal::add);
 115:     }
 116: 
 117:     // RN-08: una orden solo se recibe o se cancela mientras está PENDIENTE.
 118:     private void validarPendiente(OrdenCompra orden, String accion) {
 119:         if (orden.getEstado() != EstadoOrdenCompra.PENDIENTE) {
 120:             throw new OperacionInvalidaException(
 121:                     "No se puede " + accion + " una orden en estado " + orden.getEstado());
 122:         }
 123:     }
 124: 
 125:     private OrdenCompra buscarOrden(Long idOrden) {
 126:         return ordenes.findByIdAndIdEmpresa(idOrden, contexto.getIdEmpresa())
 127:                 .orElseThrow(() -> new RecursoNoEncontradoException(
 128:                         "Orden de compra no encontrada: " + idOrden));
 129:     }
 130: }
